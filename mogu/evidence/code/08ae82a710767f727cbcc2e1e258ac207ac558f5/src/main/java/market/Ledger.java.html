package market;

import org.springframework.stereotype.Service;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.TransactionDefinition;
import java.util.*;
import java.util.function.Supplier;
import static market.Support.*;

@Service
class Ledger {
    final JdbcTemplate db; final TransactionTemplate tx; final Support codec; final Hooks hooks; final Provider pg;
    final long holdSeconds=setting("HOLD_SECONDS",120), maxPending=setting("MAX_PENDING",256);
    Ledger(JdbcTemplate db,PlatformTransactionManager manager,Support codec,Hooks hooks,Provider pg){
        this.db=db;this.pg=pg;this.codec=codec;this.hooks=hooks;tx=new TransactionTemplate(manager);
        tx.setIsolationLevel(TransactionDefinition.ISOLATION_READ_COMMITTED);tx.setTimeout(5);
    }
    record Reply(int status,Map<String,Object> body){}
    Reply answer(int status,String code){return new Reply(status,Map.of("code",code));}
    Map<String,Object> one(String q,Object... a){var rows=db.queryForList(q,a);return rows.isEmpty()?null:rows.getFirst();}
    <T>T transaction(Supplier<T> work){return tx.execute(s->work.get());}
    Reply command(long user,String op,String scope,String key,Object payload,Supplier<Reply> work){
        require(key!=null && key.matches("[A-Za-z0-9._:-]{1,128}"),400,"IDEMPOTENCY_KEY_REQUIRED");
        String digest=hash(codec.encode(payload));
        Reply r=transaction(()->{
            db.update("INSERT INTO request_results(principal,operation,scope,idem_key,request_hash) VALUES(?,?,?,?,?) ON DUPLICATE KEY UPDATE idem_key=idem_key",user,op,scope,key,digest);
            var existing=one("SELECT * FROM request_results WHERE principal=? AND operation=? AND scope=? AND idem_key=? FOR UPDATE",user,op,scope,key);
            if(!digest.equals(existing.get("request_hash")))return answer(409,"IDEMPOTENCY_CONFLICT");
            if(existing.get("body")!=null)return new Reply(((Number)existing.get("http_status")).intValue(),codec.decode(str(existing,"body")));
            Reply result=work.get();
            db.update("UPDATE request_results SET http_status=?,body=? WHERE principal=? AND operation=? AND scope=? AND idem_key=?",result.status,codec.encode(result.body),user,op,scope,key);
            return result;
        });
        hooks.hit(op+"_after_commit");return r;
    }
    Map<String,Object> guard(long sale,long user){
        db.update("INSERT INTO user_guard(sale_id,user_id) VALUES(?,?) ON DUPLICATE KEY UPDATE user_id=user_id",sale,user);
        return one("SELECT * FROM user_guard WHERE sale_id=? AND user_id=? FOR UPDATE",sale,user);
    }
    // All mutations of an existing reservation: guard -> reservation -> attempt -> inventory -> budget.
    // Fresh reservation: request -> guard -> inventory -> INSERT new reservation. No worker can see that new ID yet.
    Map<String,Object> lockReservation(String id){
        var initial=one("SELECT sale_id,user_id FROM reservations WHERE id=?",id);
        require(initial!=null,404,"NOT_FOUND");guard(num(initial,"sale_id"),num(initial,"user_id"));
        return one("SELECT *, expires_at<=UTC_TIMESTAMP(6) AS expired FROM reservations WHERE id=? FOR UPDATE",id);
    }
    boolean expired(Map<String,Object> r){return num(r,"expired")!=0;}
    Map<String,Object> view(String id,long user){
        var r=one("SELECT * FROM reservations WHERE id=?",id);require(r!=null,404,"NOT_FOUND");require(num(r,"user_id")==user,403,"FORBIDDEN");
        var a=one("SELECT id,intent,fact,workflow,review_reason FROM attempts WHERE reservation_id=?",id);
        var result=new LinkedHashMap<>(r);if(a!=null)result.put("payment",a);return result;
    }
    Reply reserve(long user,long sale,String key,Map<String,Object> body){
        return command(user,"reserve",""+sale,key,new TreeMap<>(body),()->{
            if(!body.keySet().equals(Set.of("quantity")) || !(body.get("quantity") instanceof Number) || ((Number)body.get("quantity")).doubleValue()!=1.0)return answer(400,"QUANTITY_MUST_BE_ONE");
            var g=guard(sale,user);
            if(g.get("active_id")!=null||g.get("purchased_id")!=null)return answer(409,"USER_LIMIT");
            one("SELECT id FROM sales WHERE id=? FOR UPDATE",sale);
            var s=one("SELECT *, opens_at<=UTC_TIMESTAMP(6) AND closes_at>UTC_TIMESTAMP(6) AS is_open FROM sales WHERE id=?",sale);
            if(s==null)return answer(404,"SALE_NOT_FOUND");
            if(num(s,"is_open")==0)return answer(409,"SALE_CLOSED");
            if(db.update("UPDATE sales SET available=available-1 WHERE id=? AND available>0 AND opens_at<=UTC_TIMESTAMP(6) AND closes_at>UTC_TIMESTAMP(6)",sale)!=1)return answer(409,"SOLD_OUT");
            hooks.hit("reserve_after_stock");
            String id=id(),order="mogu_"+id.replace("-","");
            db.update("INSERT INTO reservations(id,sale_id,user_id,state,price,currency,order_id,expires_at) VALUES(?,?,?,'HELD',?,?,?,TIMESTAMPADD(SECOND,?,UTC_TIMESTAMP(6)))",id,sale,user,s.get("price"),s.get("currency"),order,holdSeconds);
            db.update("UPDATE user_guard SET active_id=? WHERE sale_id=? AND user_id=?",id,sale,user);
            return new Reply(201,Map.of("id",id,"code","RESERVED","orderId",order,"amount",s.get("price"),"currency",s.get("currency")));
        });
    }
    Reply pay(long user,String rid,String key,Map<String,Object> body){
        return command(user,"pay",rid,key,new TreeMap<>(body),()->{
            var r=lockReservation(rid);require(num(r,"user_id")==user,403,"FORBIDDEN");
            require(body.keySet().equals(Set.of("paymentKey","orderId","amount","currency")),400,"INVALID_PAYMENT_FIELDS");
            String pk=str(body,"paymentKey");require(pk.matches("[A-Za-z0-9_-]{1,200}"),400,"INVALID_PAYMENT_KEY");
            require(str(r,"order_id").equals(body.get("orderId")) && body.get("amount") instanceof Number && ((Number)body.get("amount")).doubleValue()==num(r,"price") && r.get("currency").equals(body.get("currency")),409,"PAYMENT_BINDING_MISMATCH");
            var old=one("SELECT * FROM attempts WHERE reservation_id=?",rid);
            if(old!=null)return answer(409,"ATTEMPT_ALREADY_BOUND");
            if(!str(r,"state").equals("HELD"))return answer(409,"NOT_HELD");
            if(expired(r)){release(r,"EXPIRED","HOLD_TTL");return answer(409,"EXPIRED");}
            if(db.update("UPDATE budgets SET pending=pending+1 WHERE id=1 AND pending<?",maxPending)!=1)throw new Failure(503,"PAYMENT_ADMISSION_CLOSED");
            String aid=id();
            db.update("INSERT INTO attempts(id,reservation_id,payment_key,order_id,amount,currency,merchant,credential_scope) VALUES(?,?,?,?,?,?,?,?)",aid,rid,pk,r.get("order_id"),r.get("price"),r.get("currency"),env("PG_MERCHANT","mogu-sim"),pg.scope());
            db.update("INSERT INTO operations(attempt_id,kind,idem_key) VALUES(?,'CONFIRM',?)",aid,id());
            db.update("UPDATE reservations SET state='PAYMENT_PENDING' WHERE id=?",rid);
            return new Reply(202,Map.of("code","PAYMENT_PROCESSING","id",rid,"attemptId",aid));
        });
    }
    Reply cancel(long user,String rid,String key){
        return command(user,"cancel",rid,key,Map.of("id",rid),()->{
            var r=lockReservation(rid);require(num(r,"user_id")==user,403,"FORBIDDEN");
            String state=str(r,"state");
            if(state.equals("PURCHASED"))return answer(409,"PURCHASE_ALREADY_COMMITTED");
            if(state.equals("HELD")){release(r,"CANCELLED","USER_CANCEL");return answer(200,"CANCELLED");}
            if(!state.equals("PAYMENT_PENDING"))return answer(200,state);
            var a=one("SELECT * FROM attempts WHERE reservation_id=? FOR UPDATE",rid);
            db.update("UPDATE attempts SET intent='CANCEL_REQUESTED',dirty=dirty+1,next_at=UTC_TIMESTAMP(6) WHERE id=?",a.get("id"));
            var op=one("SELECT * FROM operations WHERE attempt_id=? AND kind='CONFIRM' FOR UPDATE",a.get("id"));
            if(str(op,"dispatch").equals("PREPARED")){
                db.update("UPDATE operations SET dispatch='BLOCKED' WHERE attempt_id=? AND kind='CONFIRM'",a.get("id"));
                release(r,"CANCELLED","NEVER_ARMED");resolve(str(a,"id"),"NO_CHARGE_FINAL");return answer(200,"CANCELLED");
            }
            return answer(202,"CANCEL_REQUESTED");
        });
    }
    void release(Map<String,Object> r,String state,String reason){
        db.update("UPDATE reservations SET state=?,reason=?,terminal_at=UTC_TIMESTAMP(6) WHERE id=?",state,reason,r.get("id"));
        db.update("UPDATE sales SET available=available+1 WHERE id=?",r.get("sale_id"));
        db.update("UPDATE user_guard SET active_id=NULL WHERE sale_id=? AND user_id=? AND active_id=?",r.get("sale_id"),r.get("user_id"),r.get("id"));
    }
    void resolve(String aid,String fact){
        int n=db.update("UPDATE attempts SET fact=?,workflow='RESOLVED',resolved_at=UTC_TIMESTAMP(6),lease_until=NULL WHERE id=? AND resolved_at IS NULL",fact,aid);
        if(n==1)db.update("UPDATE budgets SET pending=pending-1 WHERE id=1");
    }
    void expire(){
        var rows=db.queryForList("SELECT id FROM reservations WHERE state='HELD' AND expires_at<=UTC_TIMESTAMP(6) ORDER BY expires_at,id LIMIT 32");
        for(var row:rows)transaction(()->{var r=lockReservation(str(row,"id"));if(str(r,"state").equals("HELD")&&expired(r))release(r,"EXPIRED","HOLD_TTL");return null;});
    }
}
