package market;

import org.springframework.stereotype.Component;
import org.springframework.scheduling.annotation.Scheduled;
import io.micrometer.core.instrument.MeterRegistry;
import jakarta.annotation.PreDestroy;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicBoolean;
import static market.Support.*;

@Component
class Recovery {
    final Ledger l; final Provider pg; final Hooks hooks; final MeterRegistry metrics;
    static int workerCount(){long n=setting("PG_WORKERS",4);require(n>=1 && n<=32,500,"INVALID_PG_WORKERS");return (int)n;}
    final int concurrency=workerCount();
    final ExecutorService workers=Executors.newFixedThreadPool(concurrency); final Semaphore slots=new Semaphore(concurrency);
    final AtomicBoolean paused=new AtomicBoolean(false);
    final long leaseMs=setting("LEASE_MS",3000),pollMs=setting("POLL_MS",500),reviewMs=setting("REVIEW_AFTER_MS",30000);
    Recovery(Ledger l,Provider p,Hooks h,MeterRegistry m){this.l=l;pg=p;hooks=h;metrics=m;m.gauge("mogu_recovery_active",slots,s->concurrency-s.availablePermits());}
    @PreDestroy void stop(){workers.shutdownNow();}
    @Scheduled(fixedDelay=50) void dispatch(){
        if(paused.get())return;
        try{
            for(int i=0;i<concurrency && slots.tryAcquire();i++){
                Map<String,Object> a;
                try{a=claim();}catch(Exception e){slots.release();throw e;}
                if(a==null){slots.release();break;}
                metrics.counter("mogu_recovery_claims").increment();
                workers.submit(()->{var sample=io.micrometer.core.instrument.Timer.start(metrics);try{work(a);}catch(Exception e){metrics.counter("mogu_recovery_errors").increment();}finally{sample.stop(metrics.timer("mogu_recovery_work"));slots.release();}});
            }
        }catch(Exception e){metrics.counter("mogu_scan_errors").increment();}
    }
    @Scheduled(fixedDelay=200) void expiry(){if(paused.get())return;try{l.expire();}catch(Exception e){metrics.counter("mogu_expiry_errors").increment();}}
    Map<String,Object> claim(){return l.transaction(()->{
        var a=l.one("SELECT * FROM attempts WHERE workflow<>'RESOLVED' AND next_at<=UTC_TIMESTAMP(6) AND (lease_until IS NULL OR lease_until<=UTC_TIMESTAMP(6)) ORDER BY next_at,id LIMIT 1 FOR UPDATE SKIP LOCKED");
        if(a==null)return null;
        l.db.update("UPDATE attempts SET generation=generation+1,lease_until=TIMESTAMPADD(MICROSECOND,?,UTC_TIMESTAMP(6)) WHERE id=?",leaseMs*1000,a.get("id"));
        a.put("generation",num(a,"generation")+1);return a;
    });}
    Map<String,Object> locked(Map<String,Object> claimed){
        l.lockReservation(str(claimed,"reservation_id"));
        return l.one("SELECT *, TIMESTAMPDIFF(MICROSECOND,started_at,UTC_TIMESTAMP(6))/1000 AS age_ms FROM attempts WHERE id=? FOR UPDATE",claimed.get("id"));
    }
    boolean owns(Map<String,Object> current,Map<String,Object> claim){return num(current,"generation")==num(claim,"generation");}
    String arm(Map<String,Object> claim,String kind){return l.transaction(()->{
        var a=locked(claim);if(!owns(a,claim)||str(a,"workflow").equals("RESOLVED"))return null;
        if(!str(a,"credential_scope").equals(pg.scope())){review(str(a,"id"),"CREDENTIAL_SCOPE_CHANGED");return null;}
        if(kind.equals("CONFIRM")&&!str(a,"intent").equals("CONTINUE"))return null;
        var op=l.one("SELECT *, IFNULL(TIMESTAMPDIFF(SECOND,armed_at,UTC_TIMESTAMP(6)),0) AS age_seconds FROM operations WHERE attempt_id=? AND kind=? FOR UPDATE",a.get("id"),kind);
        if(op==null)return null;
        if(str(op,"dispatch").equals("BLOCKED"))return null;
        if(num(op,"age_seconds")>=setting("PG_IDEMPOTENCY_SECONDS",1296000)) {review(str(a,"id"),"IDEMPOTENCY_WINDOW_EXPIRED");return null;}
        l.db.update("UPDATE operations SET dispatch='ARMED',armed_at=COALESCE(armed_at,UTC_TIMESTAMP(6)),calls=calls+1 WHERE attempt_id=? AND kind=?",a.get("id"),kind);
        return str(op,"idem_key");
    });}
    void review(String aid,String reason){l.db.update("UPDATE attempts SET workflow='REVIEW',review_reason=? WHERE id=?",reason,aid);}
    Map<String,Object> call(String kind,Map<String,Object> a,String key){
        metrics.counter("mogu_pg_calls","kind",kind).increment();
        var sample=io.micrometer.core.instrument.Timer.start(metrics);
        try{return pg.request(kind,a,key);}finally{sample.stop(metrics.timer("mogu_pg_duration","kind",kind));}
    }
    void work(Map<String,Object> a){
        hooks.hit("worker_claimed");
        var op=l.one("SELECT * FROM operations WHERE attempt_id=? AND kind='CONFIRM'",a.get("id"));
        Map<String,Object> observation;
        String source;
        if(str(op,"dispatch").equals("PREPARED")){
            String key=arm(a,"CONFIRM");if(key==null){finish(a);return;}
            hooks.hit("confirm_armed");observation=call("CONFIRM",a,key);source="CONFIRM";
        }else{observation=call("QUERY",a,"");source="QUERY";}
        hooks.hit("pg_observed");
        apply(a,observation,source);
        // A query of READY/NOT_FOUND is not a final failure. Retrying CONTINUE uses the SAME operation.
        if(source.equals("QUERY") && Set.of("READY","UNKNOWN","IN_PROGRESS").contains(str(observation,"status"))){
            String key=arm(a,"CONFIRM");
            if(key!=null){hooks.hit("confirm_armed");apply(a,call("CONFIRM",a,key),"CONFIRM");}
        }
        var current=l.one("SELECT * FROM attempts WHERE id=?",a.get("id"));
        if(str(current,"intent").equals("CANCEL_REQUESTED")&&str(current,"fact").equals("APPROVED")&&!str(current,"workflow").equals("RESOLVED")){
            String key=arm(a,"CANCEL");if(key!=null){hooks.hit("cancel_armed");apply(a,call("CANCEL",a,key),"CANCEL");}
        }
        hooks.hit("worker_before_finish");finish(a);
    }
    void apply(Map<String,Object> claim,Map<String,Object> observed,String source){
        l.transaction(()->{
            var a=locked(claim);String aid=str(a,"id");
            String status=str(observed,"status");String fact=switch(status){case "DONE"->"APPROVED";case "CANCELED"->"CANCELED_FINAL";case "ABORTED","EXPIRED"->"NO_CHARGE_FINAL";default->"UNOBSERVED";};
            boolean valid=fact.equals("UNOBSERVED") || (str(observed,"paymentKey").equals(str(a,"payment_key"))&&str(observed,"orderId").equals(str(a,"order_id"))&&observed.get("totalAmount") instanceof Number&&num(observed,"totalAmount")==num(a,"amount")&&str(observed,"currency").equals(str(a,"currency"))&&str(observed,"mid").equals(str(a,"merchant"))&&Set.of("CARD","카드").contains(str(observed,"method")));
            if(status.equals("CANCELED") && (!(observed.get("balanceAmount") instanceof Number)||num(observed,"balanceAmount")!=0))valid=false;
            if(!Set.of("DONE","CANCELED","ABORTED","EXPIRED","READY","IN_PROGRESS","UNKNOWN").contains(status))valid=false;
            // Actual provider final no-charge states require provider-specific future-dispatch proof.
            if(fact.equals("NO_CHARGE_FINAL")&&!pg.base.startsWith("http://127.0.0.1:"))valid=false;
            l.db.update("INSERT IGNORE INTO observations(attempt_id,source,fact,valid,detail) VALUES(?,?,?,?,?)",aid,source,fact,valid,status);
            if(!owns(a,claim)){l.db.update("UPDATE attempts SET dirty=dirty+1,next_at=UTC_TIMESTAMP(6) WHERE id=?",aid);return null;}
            if(!valid){review(aid,"OBSERVATION_BINDING_OR_CONTRACT");return null;}
            var r=l.one("SELECT * FROM reservations WHERE id=?",a.get("reservation_id"));
            String state=str(r,"state");
            if(!state.equals("PAYMENT_PENDING")){
                if(state.equals("PURCHASED")&&!fact.equals("APPROVED")&&!fact.equals("UNOBSERVED"))review(aid,"POST_PURCHASE_EXTERNAL_CHANGE");
                else if(!fact.equals("UNOBSERVED"))l.db.update("UPDATE attempts SET workflow='RESOLVED',lease_until=NULL WHERE id=?",aid);
                return null;
            }
            if(fact.equals("UNOBSERVED"))return null;
            if(str(a,"fact").equals("CANCELED_FINAL"))return null;
            if(fact.equals("NO_CHARGE_FINAL")&&str(a,"fact").equals("APPROVED")){review(aid,"CONFLICTING_FINAL_FACT");return null;}
            l.db.update("UPDATE attempts SET fact=? WHERE id=?",fact,aid);
            if(fact.equals("APPROVED")){
                if(str(a,"intent").equals("CONTINUE")){
                    hooks.hit("purchase_before_commit");
                    l.db.update("UPDATE reservations SET state='PURCHASED',terminal_at=UTC_TIMESTAMP(6) WHERE id=?",r.get("id"));
                    l.db.update("UPDATE user_guard SET active_id=NULL,purchased_id=? WHERE sale_id=? AND user_id=? AND active_id=?",r.get("id"),r.get("sale_id"),r.get("user_id"),r.get("id"));
                    l.resolve(aid,fact);metrics.counter("mogu_purchases","source",source).increment();
                }else l.db.update("INSERT IGNORE INTO operations(attempt_id,kind,idem_key) VALUES(?,'CANCEL',?)",aid,id());
            }else{
                l.release(r,fact.equals("CANCELED_FINAL")||str(a,"intent").equals("CANCEL_REQUESTED")?"CANCELLED":"EXPIRED",fact);
                l.db.update("UPDATE operations SET dispatch='BLOCKED' WHERE attempt_id=? AND dispatch='PREPARED'",aid);
                l.resolve(aid,fact);
            }
            return null;
        });
        hooks.hit("purchase_after_commit");
    }
    void finish(Map<String,Object> claim){l.transaction(()->{
        var a=locked(claim);if(!owns(a,claim))return null;
        if(!str(a,"workflow").equals("RESOLVED")){
            boolean isReview=str(a,"workflow").equals("REVIEW") || num(a,"age_ms")>=reviewMs;
            long delay=isReview?setting("REVIEW_POLL_MS",2000):Math.min(pollMs*4,pollMs*(1L<<Math.min(2,num(a,"rounds"))));
            // dirty generation preserves a signal arriving during execution; floor prevents unbounded webhook amplification.
            if(num(a,"dirty")!=num(claim,"dirty"))delay=Math.min(delay,pollMs);
            l.db.update("UPDATE attempts SET workflow=?,review_reason=CASE WHEN ? THEN COALESCE(review_reason,'LONG_UNKNOWN') ELSE review_reason END,next_at=TIMESTAMPADD(MICROSECOND,?,UTC_TIMESTAMP(6)),rounds=rounds+1,lease_until=NULL WHERE id=?",isReview?"REVIEW":"RECONCILING",isReview,delay*1000,a.get("id"));
        }else l.db.update("UPDATE attempts SET lease_until=NULL WHERE id=?",a.get("id"));
        l.db.update("UPDATE inbox SET processed=TRUE WHERE attempt_id=? AND processed=FALSE",a.get("id"));
        return null;
    });}
}
