"""Derive corrected occupancy from immutable prior runs, never replace their raw data."""
import json,datetime,hashlib
from pathlib import Path
root=Path(__file__).resolve().parents[1]/'evidence'
def epoch(s):return datetime.datetime.fromisoformat(s).replace(tzinfo=datetime.timezone.utc).timestamp()
for folder in sorted(root.glob('*recovery-v[123]-*')):
    raw=folder/'results.json'
    if not raw.exists():continue
    result=json.loads(raw.read_text())['result'];db=json.loads((folder/'db.json').read_text())
    reservations={r['id']:r for r in db['reservations']};attempts={a['reservation_id']:a for a in db['attempts']}
    rows=[]
    for r in result['rows']:
        reservation=reservations[r['id']];a=attempts[r['id']]
        end=epoch(reservation['terminal_at']);start=epoch(reservation['created_at']);pending=epoch(a['started_at'])
        rows.append(dict(id=r['id'],reserved_at=start,payment_started_at=pending,terminal_at=end,allocation_to_resolution_seconds=end-start,payment_pending_seconds=end-pending,accepted_response_to_resolution_seconds=end-r['accepted']))
    value={'definition':'Unresolved stock occupancy: reservation DB created_at to terminal_at; purchased stock remains consumed after this interval. Measured rows only; warmup excluded. UTC microsecond timestamps.', 'unit':'item.seconds','source_sha256':{n:hashlib.sha256((folder/n).read_bytes()).hexdigest() for n in ['db.json','results.json']},'original_stock_seconds':result['stock_seconds'],'allocation_stock_seconds':sum(r['allocation_to_resolution_seconds'] for r in rows),'pending_stock_seconds':sum(r['payment_pending_seconds'] for r in rows),'rows':rows}
    (folder/'occupancy-corrected.json').write_text(json.dumps(value,indent=2))
    print(folder.name,round(value['original_stock_seconds'],3),round(value['allocation_stock_seconds'],3))
